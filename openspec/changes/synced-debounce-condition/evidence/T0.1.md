func newProbeEnv(prefix string) *probeEnv {
    suffix := testNameSuffix()
    e := &probeEnv{
        ns:       "default",
        invName:  prefix + "-inv-" + suffix,
        sinkName: prefix + "-pg-" + suffix,
        store:    collect.NewStore(),
        recorder: &recordingBackend{},
        fail:     &atomic.Bool{},
    }
    e.store.Upsert(collect.Item{
        TargetNamespace: e.ns, TargetName: "demo-target", UID: "uid-" + prefix,
        Namespace: e.ns, Name: "nginx", Version: "v1", Kind: "Deployment",
    })
    sinkObj, secret := createPostgresSinkFixtures(e.sinkName, "pg-"+e.sinkName, e.ns)
    Expect(k8sClient.Create(ctx, sinkObj)).To(Succeed())
    DeferCleanup(func() { _ = k8sClient.Delete(ctx, sinkObj) })
    Expect(k8sClient.Create(ctx, secret)).To(Succeed())
    DeferCleanup(func() { _ = k8sClient.Delete(ctx, secret) })
    e.sinkObj = sinkObj

    inv := &kollectdevv1alpha1.KollectInventory{
        ObjectMeta: metav1.ObjectMeta{Name: e.invName, Namespace: e.ns},

            DatabaseSinkRefs:  kollectdevv1alpha1.NewSinkRefList(e.sinkName),
        },
    }
    Expect(k8sClient.Create(ctx, inv)).To(Succeed())
    DeferCleanup(func() { _ = k8sClient.Delete(ctx, inv) })

    exportErr := kollecterrors.Transient(errors.New("sink unavailable"))
    reg := newPostgresRecordingRegistryWithSelector(func(_ kollectdevv1alpha1.KollectSinkSpec) (sink.Backend, error) {
        if e.fail.Load() {

    return e
}
