## Probe sources

Evidence under `openspec/` is Markdown only. The guard's hand mutants, run as
`bash /tmp/g9/gate-mutants.sh <repo>`
on `852709f11`:

```bash
#!/usr/bin/env bash
# Hand mutants of hack/test/openspec_evidence_markdown_only_test.sh (G9): each must make
# --self-test exit non-zero. Restores the gate from the first saved copy on exit.
# usage: bash gate-mutants.sh <repo>
set -uo pipefail
cd "${1:?repo}" || exit 1
G=hack/test/openspec_evidence_markdown_only_test.sh
SAVE="$(mktemp)"
cp "$G" "$SAVE"
trap 'cp "$SAVE" "$G"; rm -f "$SAVE"' EXIT
killed=0 survived=0 notlanded=0
run() {
  local label="$1" old="$2" new="$3" out rc
  cp "$SAVE" "$G"
  if ! python3 -I -c 'import sys; p,o,n=sys.argv[1:]; s=open(p).read(); assert o in s, o; open(p,"w").write(s.replace(o,n,1))' "$G" "$old" "$new"; then
    echo "MUTANT-NOT-LANDED $label"; notlanded=$((notlanded + 1)); return
  fi
  out="$(bash "$G" --self-test 2>&1)"; rc=$?
  if [[ "$rc" -ne 0 ]]; then killed=$((killed + 1)); else survived=$((survived + 1)); fi
  echo "$label exit=$rc :: $(grep FAIL <<<"$out" | head -1)"
}
run "suffix check removed" 'if [[ "${f}" != *.md ]]; then' 'if false; then'
run "suffix check loosened" 'if [[ "${f}" != *.md ]]; then' 'if [[ "${f}" != *.md* ]]; then'
run "archive not scanned" "EVIDENCE_GLOB=':(glob)openspec/changes/**/evidence/**'" "EVIDENCE_GLOB=':(glob)openspec/changes/[!a]*/evidence/**'"
run "nested files not scanned" "EVIDENCE_GLOB=':(glob)openspec/changes/**/evidence/**'" "EVIDENCE_GLOB=':(glob)openspec/changes/**/evidence/*'"
run "empty scan passes" 'if [[ "${#files[@]}" -eq 0 ]]; then' 'if false; then'
run "first offender only" 'bad=$((bad + 1))' 'return 1'
run "offenders counted but pass" '[[ "${bad}" -eq 0 ]] || return 1' 'true'
run "top-level check removed" 'if [[ "${top}" != "${root}" ]]; then' 'if false; then'
run "not-a-repo passes" '    echo "FAIL: ${root} is not a git work tree; nothing was scanned" >&2
    return 1' '    return 0'
cp "$SAVE" "$G"
echo "killed=$killed survived=$survived not-landed=$notlanded"
bash "$G" >/dev/null || { echo "FAIL: the restored gate does not pass"; exit 1; }
cmp -s "$G" "$SAVE" && echo "restored gate passes, identical to first copy"
[[ "$survived" -eq 0 && "$notlanded" -eq 0 ]] || { echo "FAIL: survived or not-landed mutants"; exit 1; }
```
