## What the guard does

`hack/test/readme_quickstart_test.sh [--self-test] [<tree-root>]`. Plain mode reads `README.md`,
`Taskfile.yml` (and the Taskfiles it includes, recursively), and `config/crd/bases/*.yaml` of the
tree. It takes the first `## Quick start…` section (to the next `#`/`##` heading outside a fence),
keeps the non-blank lines of its `sh` fences, splits each at shell separators (`&&`, `||`, `;`,
`|`, `&`), at `(`, `)` and backticks, and at a `#` that starts a word (so `# or: task dev-up` is a
segment of its own), and checks the first `task` or `kubectl` word of each segment. Failure lines
are `FAIL: <clause>: …` with clauses `section`, `extract`, `task`, `kubectl`, `apply`, `get`,
`input`; all violations are reported before exit 1. Runtime: plain 0 s, `--self-test` 3 s; no
Docker, no cluster, no `task` binary. The split is done in awk (portable `\n`), not GNU sed.
A line ending in `\` is joined with the next; an indented or `~~~` fence opener in the section
fails `extract` (it would hide its commands from this reader).

Not duplicated: `demo_task_aliases_test.sh` checks that `demo-up`/`demo-down` exist and delegate
to the hero tasks; this guard checks only that README-named tasks exist (any task, by name), no
delegation (held by review).

#!/usr/bin/env python3
"""Hand mutants of readme_quickstart_test.sh's check code; each must make --self-test exit 1."""
import os, shutil, subprocess, sys, tempfile

repo = sys.argv[1]
src = open(os.path.join(repo, "hack/test/readme_quickstart_test.sh")).read()
M = [
    ("no comment split", 'gsub(/(^|[[:space:]])#/, "\\n"); ', ""),
    ("no separator split", ' gsub(/&&|\\|\\||[;|&()`]/, "\\n");', ""),
    ("no single-& split", "[;|&()`]/", "[;|()`]/"),
    ("no bracket split", "[;|&()`]/", "[;|&`]/"),
    ("flag before verb accepted", 'if [[ "${w[i + 1]:-}" == -* ]]; then', 'if false; then'),
    ("unclosed rule only for sh", '    infence && /^## / {', '    infence && lang == "sh" && /^## / {'),
    ("EOF unclosed accepted", 'else if (infence) print "UNCLOSED\\t" open', ''),
    ("no continuation join", '    infence && lang == "sh" && /\\\\$/ {', '    infence && lang == "never" && /\\\\$/ {'),
    ("indented fence accepted", '/^[[:space:]]+(```|~~~)|^~~~/ { print "UNSUPPORTED', '/^NEVER/ { print "UNSUPPORTED'),
    ("continued last line dropped", 'if (buf != "") print at "\\t" buf; ', ''),
    ("substring task match", 'grep -Fx -- "${w[j]}" <<<"${tasks}"', 'grep -F -- "${w[j]}" <<<"${tasks}"'),
    ("substring name match", 'grep -Fx -- "${res}" <<<"${names}"', 'grep -F -- "${res}" <<<"${names}"'),
    ("no comma split", "IFS=',' read -ra r <<<\"${res}\"", 'r=("${res}")'),
    ("any fence language", 'if (lang == "sh" && line ~', 'if (line ~'),
    ("section never ends", "    /^##?[[:space:]]/ { exit }\n", ""),
    ("no include recursion", '    taskfile_tasks "${path}" "${prefix}${ns}:"\n', "    :\n"),
    ("no prefix", 'taskfile_tasks "${path}" "${prefix}${ns}:"', 'taskfile_tasks "${path}" ""'),
    ("apply path unchecked", '[[ -n "${path}" && -e "${ROOT}/${path}" ]] ||', "true ||"),
    ("empty extraction accepted", '"") die extract', '"__never__") die extract'),
    ("missing section accepted", "SECTION-MISSING) die section", "__never__) die section"),
    ("unclosed accepted", "*UNCLOSED*) die extract", "__never__) die extract"),
    ("heading inside fence ignored", '    infence && /^## / { print "UNCLOSED\\t" open; infence = 0; exit }\n', ""),
    ("shortNames not read", '    inshort && /^    - / { v = $0; sub(/^    - /, "", v); print v; next }\n', ""),
    ("plural/singular not read", '/^    (plural|singular):/', '/^    (nosuch):/'),
    ("tasks block not read", 'intasks = ($0 ~ /^tasks:[[:space:]]*$/)', 'intasks = 0'),
    ("bad() does not fail", "  FAILED=1\n}", "  :\n}"),
    ("missing Taskfile accepted", '[[ -f "${file}" ]] || die input', '[[ -f "${file}" ]] || return 0; die input'),
    ("CRD absence accepted", '[[ -f "${crds[0]}" ]] || die input "no CRD', 'return 0; die input "no CRD'),
]
bad = 0
for label, old, new in M:

        print(f"ANCHOR {label}: count={src.count(old)}"); bad += 1; continue
    t = tempfile.mkdtemp()
    for f in ("README.md", "Taskfile.yml"):
        shutil.copy(os.path.join(repo, f), t)
    shutil.copytree(os.path.join(repo, "hack/task"), os.path.join(t, "hack/task"))

    r = subprocess.run(["bash", os.path.join(t, "hack/test/g.sh"), "--self-test"], capture_output=True, text=True)
    plain = subprocess.run(["bash", os.path.join(t, "hack/test/g.sh")], capture_output=True, text=True)

    status = "KILLED" if r.returncode != 0 else "SURVIVED"
    if r.returncode == 0: bad += 1
    print(f"{status} {label}: self-test rc={r.returncode} plain rc={plain.returncode}; by: {'; '.join(k.split(':')[2].strip() for k in killed)[:150]}")

## Learnings

- class: sensor. Reusable: a guard that reads Markdown fences must treat a fence that runs into
  the next `##` heading as unclosed; a later bare fence otherwise "closes" it and the section
  boundary is lost (G3).
- class: sensor. Reusable: a hand mutant that drops a tokenizer step can survive every
  task-named case; add one case with two commands on a line (G3).
